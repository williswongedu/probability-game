    const SUPABASE_ANON_KEY = 'eyJhbGciOiJIUzI1NiIsInR5cCI6IkpXVCJ9.eyJpc3MiOiJzdXBhYmFzZSIsInJlZiI6Imhqc2N5bWZvbHV1cXNsdGZzaHh1Iiwicm9sZSI6ImFub24iLCJpYXQiOjE3ODM1NzQxNzcsImV4cCI6MjA5OTE1MDE3N30.AVaR1LWvnaONa5KIYr4rTteqZ-VgdPOvFgGvLw8f9Zk';

    const studentId = localStorage.getItem('supabase_student_id');
    let gems = 1000;
    let isBoosted = false; // 是否獲得爆率加成

    // 動態 DSE 題目生成
    let currentBoosterAns = 0;
    const gachaPool = [
        { name: '雷鳴精算師 · 高斯', attr: '雷', stars: '⭐⭐⭐⭐⭐', emoji: '⚡', skill: '常態修正', desc: '概率戰中答對時直接鎖定滿額輸出！' },

        { name: '鋼鐵防線 · 巴斯卡', attr: '土', stars: '⭐⭐⭐⭐', emoji: '🛡️', skill: '三角防護', desc: '盾牌防守失敗時提供一次公式精算重骰機會。' },
        { name: '幻影刺客 · 貝葉斯', attr: '暗', stars: '⭐⭐⭐⭐', emoji: '🗡️', skill: '天雷神罰', desc: '在特定潮濕狀態下，造成翻倍暴擊傷害！' }
    ];

    // 初始化與加載寶石
    async function initGacha() {
        if (!studentId) {
            alert('請先在學院傳送門登入！');
            window.location.href = 'index.html';
            return;
        }

        const { data } = await supabaseClient.from('student_progress').select('gems').eq('student_id', studentId).single();
        if (data) {
            gems = data.gems;
            document.getElementById('gemCount').innerText = gems;
        }
        generateBoosterQuestion();
    }

    // 生成隨機 DSE 概率題目 (Plan B)
    function generateBoosterQuestion() {
        const pulls = Math.floor(Math.random() * 2) + 2; // 2 或 3 次
        const ratePercent = Math.floor(Math.random() * 3) * 5 + 10; // 10%, 15%, 或 20%
        const rateDecimal = ratePercent / 100;
        
        // P(至少中一次) = 1 - (1 - rate)^pulls
        const rawAns = 1 - Math.pow(1 - rateDecimal, pulls);
        currentBoosterAns = parseFloat((rawAns * 100).toFixed(1)); // 保留一位小數

        document.getElementById('boosterQuest').innerText = `「召喚池5星英靈中獎率為 ${ratePercent}%。若你進行連續 ${pulls} 次單抽，請問【至少抽中一次】5星英靈的概率是多少%？」(四捨五入至小數點後一位)`;
        document.getElementById('boosterInput').value = '';
        document.getElementById('boosterFeedback').innerText = '';
    }

    // 驗證 Plan B
    function checkBooster() {
        const studentAns = parseFloat(document.getElementById('boosterInput').value);
        const fb = document.getElementById('boosterFeedback');

        if (isNaN(studentAns)) {
            fb.innerText = '⚠️ 請輸入答案！';
            fb.style.color = '#ff4d4d';
            return;
        }

        if (Math.abs(studentAns - currentBoosterAns) <= 0.2) {
            isBoosted = true;
            fb.innerText = '🔥 答對了！魔法陣爆發極致金光，5星爆率提升至 20%！';
            fb.style.color = '#00ffcc';
            document.getElementById('altar').classList.add('boosted');
        } else {
            isBoosted = false;
            fb.innerText = `❌ 算錯了，法陣維持冷色（提示：使用餘事公式 $1 - (1 - p)^n$ 算出為 ${currentBoosterAns}%）`;
            fb.style.color = '#ff4d4d';
            document.getElementById('altar').classList.remove('boosted');
        }
    }

    // 神魔之塔拉石碑觸摸滑動手勢
    const stone = document.getElementById('stone');
    const altar = document.getElementById('altar');
    let startY = 0, isDragging = false;

    stone.addEventListener('mousedown', (e) => {
        startY = e.clientY;
        isDragging = true;
    });

    window.addEventListener('mousemove', (e) => {
        if (!isDragging) return;
        let deltaY = e.clientY - startY;
        if (deltaY > 0 && deltaY < 120) {
            stone.style.transform = `translateY(${deltaY}px)`;
        }
        if (deltaY >= 110) {
            // 拉到底部，觸發召喚
            isDragging = false;
            stone.style.transform = 'translateY(0)';
            triggerSummon();
        }
    });

    window.addEventListener('mouseup', () => {
        if (isDragging) {
            isDragging = false;
            stone.style.transform = 'translateY(0)';
        }
    });

    // 執行召喚
    async function triggerSummon() {
        if (gems < 100) {
            alert('😭 你的寶石不足！請前往決鬥場答題獲取金幣。');
            return;
        }

        gems -= 100;
        document.getElementById('gemCount').innerText = gems;

        // 雲端實時更新寶石扣除
        await supabaseClient.from('student_progress').update({ gems }).eq('student_id', studentId);

        // 機率計算
        let roll = Math.random();
        let rate5Star = isBoosted ? 0.20 : 0.10; // 爆率提升

        let selectedHero = null;
        if (roll < rate5Star) {
            // 5星卡
            const fiveStars = gachaPool.filter(h => h.stars === '⭐⭐⭐⭐⭐');
            selectedHero = fiveStars[Math.floor(Math.random() * fiveStars.length)];
        } else {
            // 4星卡
            const fourStars = gachaPool.filter(h => h.stars === '⭐⭐⭐⭐');
            selectedHero = fourStars[Math.floor(Math.random() * fourStars.length)];
        }

        // 展示神魔 TCG 翻開卡牌特效
        showCardReveal(selectedHero);

        // 寫入學生的已解鎖卡牌背包
        const { data } = await supabaseClient.from('student_progress').select('unlocked_cards').eq('student_id', studentId).single();
        let cards = data.unlocked_cards || [];
        if (!cards.includes(selectedHero.name)) {
            cards.push(selectedHero.name);
            await supabaseClient.from('student_progress').update({ unlocked_cards: cards }).eq('student_id', studentId);
        }

        // 重新刷新 Plan B
        isBoosted = false;
        document.getElementById('altar').classList.remove('boosted');
        generateBoosterQuestion();
    }

    function showCardReveal(hero) {
        document.getElementById('cardName').innerText = hero.name;
        document.getElementById('cardAttr').innerText = hero.attr;
        document.getElementById('cardStars').innerText = hero.stars;
        document.getElementById('cardArt').innerText = hero.emoji;
        document.getElementById('cardSkillTitle').innerText = `【技能】${hero.skill}`;
        document.getElementById('cardSkillDesc').innerText = hero.desc;

        document.getElementById('resultOverlay').style.display = 'flex';
    }

    function closeOverlay() {
        document.getElementById('resultOverlay').style.display = 'none';
    }

    // 鼠標懸停卡牌視差折射
    const card = document.getElementById('resultCard');
    card.addEventListener('mousemove', (e) => {
        const rect = card.getBoundingClientRect();
        const x = e.clientX - rect.left;
        const y = e.clientY - rect.top;
        const rotateY = -(rect.width/2 - x) / (rect.width / 20);
        const rotateX = (rect.height/2 - y) / (rect.height / 20);
        card.style.transform = `rotateX(${rotateX}deg) rotateY(${rotateY}deg)`;
    });
    card.addEventListener('mouseleave', () => card.style.transform = 'rotateX(0) rotateY(0)');

    initGacha();
</script>
