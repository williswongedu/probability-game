<script>

    // 初始化 Supabase 客戶端
    const supabaseClient = supabase.createClient(SUPABASE_URL, SUPABASE_ANON_KEY);

    const statusMsg = document.getElementById('statusMsg');

    /**
     * 1. 學生註冊邏輯
     */
    async function handleRegister() {
        const username = document.getElementById('username').value.trim().toUpperCase();
        const password = document.getElementById('password').value;

        if (!username || !password) {
            showMsg('❌ 請輸入學號及密碼！', '#ff4d4d');
            return;
        }

        // 正則表達式驗證格式：例如 5A01 至 5F99
        const namePattern = /^[1-6][A-F][0-9]{2}$/;
        if (!namePattern.test(username)) {
            showMsg('❌ 格式錯誤！請使用「班別+兩位座號」格式，例如: 5A01', '#ff4d4d');
            return;
        }

        showMsg('🌀 正在啟動召喚儀式...', '#ffd700');

        const className = username.substring(0, 2); // 5A
        const seatNum = parseInt(username.substring(2, 4)); // 1

        try {
            // A. 向 students 表插入數據
            const { data: student, error: studentError } = await supabaseClient
                .from('students')
                .insert([{ username, password_hash: password, class: className, seat_number: seatNum }])
                .select()
                .single();

            if (studentError) {
                if (studentError.code === '23505') {
                    throw new Error('此學號已被註冊！請直接登入。');
                }
                throw studentError;
            }

            // B. 為新註冊學生創建遊戲進度，贈送 1000 Gems (寶石)
            const { error: progressError } = await supabaseClient
                .from('student_progress')
                .insert([{ student_id: student.id, gems: 1000 }]);

            if (progressError) throw progressError;

            showMsg('🎉 註冊成功！已贈送 1000 開局寶石！請點擊「登入」開始決鬥。', '#00ffcc');
        } catch (error) {
            showMsg(`❌ 註冊失敗: ${error.message}`, '#ff4d4d');
        }
    }

    /**
     * 2. 學生登入邏輯（已升級：儲存緩存並跳轉大廳）
     */
    async function handleLogin() {
        const username = document.getElementById('username').value.trim().toUpperCase();
        const password = document.getElementById('password').value;

        if (!username || !password) {
            showMsg('❌ 請輸入學號及密碼！', '#ff4d4d');
            return;
        }

        showMsg('🔑 驗證魔法身份中...', '#ffd700');

        try {
            // A. 驗證學號與密碼
            const { data: student, error: studentError } = await supabaseClient
                .from('students')
                .select('*')
                .eq('username', username)
                .eq('password_hash', password)
                .single();

            if (studentError || !student) {
                throw new Error('學號或密碼不正確！');
            }

            // B. 讀取學生的遊戲金幣存檔
            const { data: progress, error: progressError } = await supabaseClient
                .from('student_progress')
                .select('gems')
                .eq('student_id', student.id)
                .single();

            if (progressError) throw progressError;

            // 💡 關鍵升級：將學生的登入資訊存入 LocalStorage，讓後續的 gacha.html 和 battle.html 能夠讀取到他是誰
            localStorage.setItem('supabase_student_id', student.id);
            localStorage.setItem('supabase_student_name', username);

            showMsg(`🔓 歡迎回來，${username}！認證成功，正在傳送至夢幻城堡...`, '#00ffcc');
            
            // ⏳ 1.5 秒後自動跳轉到城堡大廳 (lobby.html)
            setTimeout(() => { 
                window.location.href = 'lobby.html'; 
            }, 1500);

        } catch (error) {
            showMsg(`❌ 登入失敗: ${error.message}`, '#ff4d4d');
        }
    }

    function showMsg(text, color) {
        statusMsg.innerText = text;
        statusMsg.style.color = color;
    }
