    // 遊戲初始化
    function initGame() {
        // 隨機抽選一題
        currentQuest = quests[Math.floor(Math.random() * quests.length)];
        document.getElementById('questDifficulty').innerText = currentQuest.difficulty;
        document.getElementById('questText').innerText = currentQuest.text;
        document.getElementById('answerInput').value = '';
        document.getElementById('resultFeedback').innerText = '';
        document.getElementById('skillTooltip').style.display = 'none';
        document.getElementById('skillText').innerText = currentQuest.hint;
    }

    // 點擊水晶的娛樂反饋（播放魔幻微晃動，增加趣味性）
    function interactCrystal(name) {
        const cauldron = document.getElementById('cauldron');
        cauldron.style.transform = 'scale(1.1)';
        setTimeout(() => cauldron.style.transform = 'scale(1)', 150);
        document.getElementById('resultFeedback').innerText = `✨ 已點擊 ${name} 色水晶，在腦中構思組合吧！`;
        document.getElementById('resultFeedback').style.color = '#a291c9';
    }

    // 切換英靈技能提示顯示
    function toggleSkill() {
        const tooltip = document.getElementById('skillTooltip');
        tooltip.style.display = tooltip.style.display === 'block' ? 'none' : 'block';
    }

    // 提交答案
    async function submitAnswer() {
        const studentAns = parseInt(document.getElementById('answerInput').value);
        const feedback = document.getElementById('resultFeedback');

        if (isNaN(studentAns)) {
            feedback.innerText = '⚠️ 請輸入你的答案！';
            feedback.style.color = '#ff4d4d';
            return;
        }

        if (studentAns === currentQuest.answer) {
            feedback.innerText = '🎉 融合成功！你算出了完美的元素平衡，獲得 50 寶石！';
            feedback.style.color = '#00ffcc';

            // 實時將成績儲存到 Supabase 雲端
            await updateProgressOnCloud(true);
        } else {
            feedback.innerText = '❌ 元素爆炸！比例不平衡，請檢索英靈提示，再算一次！';
            feedback.style.color = '#ff4d4d';
            
            await updateProgressOnCloud(false);
        }
    }

    /**
     * 雲端存檔更新函數
     */
    async function updateProgressOnCloud(isCorrect) {
        // 1. 如果 studentId 為空（例如非從 login 登入，僅本地調試），我們使用測試學生
        if (!studentId) {
            // 先在控制台打印提示
            console.warn('⚠️ 未檢測到學生 ID，將以測試學號進行寫入');
            // 嘗試獲取數據庫中現有第一個學生的 ID 作為測試
            const { data } = await supabaseClient.from('students').select('id').limit(1);
            if (data && data.length > 0) {
                studentId = data[0].id;
            } else {
                return; // 如果數據庫連學生都沒有，則無法寫入
            }
        }

        try {
            // A. 更新學生的寶石 (Gems)：答對加 50 寶石
            if (isCorrect) {
                // 讀取當前寶石
                const { data: progress } = await supabaseClient
                    .from('student_progress')
                    .select('gems')
                    .eq('student_id', studentId)
                    .single();
                
                if (progress) {
                    await supabaseClient
                        .from('student_progress')
                        .update({ gems: progress.gems + 50 })
                        .eq('student_id', studentId);
                }
            }

            // B. 更新 DSE 表現記錄表格 (dse_records)
            // 先嘗試讀取現有記錄
            const { data: record } = await supabaseClient
                .from('dse_records')
                .select('*')
                .eq('student_id', studentId)
                .eq('topic', 'Permutations_Combinations')
                .single();

            if (record) {
                // 如果已有記錄，累加次數
                await supabaseClient
                    .from('dse_records')
                    .update({
                        total_attempts: record.total_attempts + 1,
                        correct_attempts: record.correct_attempts + (isCorrect ? 1 : 0)
                    })
                    .eq('id', record.id);
            } else {
                // 如果是第一次答該單元，則新建一行
                await supabaseClient
                    .from('dse_records')
                    .insert([{
                        student_id: studentId,
                        topic: 'Permutations_Combinations',
                        total_attempts: 1,
                        correct_attempts: isCorrect ? 1 : 0
                    }]);
            }
            console.log('☁️ 雲端學生成績同步成功！');
        } catch (e) {
            console.error('❌ 雲端同步出錯：', e);
        }
    }

    initGame();
