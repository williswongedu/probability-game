<script>
    // ==========================================
    // ⚠️ 請在此處填入你的 Supabase 連接資訊 ⚠️

    const SUPABASE_URL = 'https://hjscymfoluuqsltfshxu.supabase.coL'; 
    const SUPABASE_ANON_KEY = 'eyJhbGciOiJIUzI1NiIsInR5cCI6IkpXVCJ9.eyJpc3MiOiJzdXBhYmFzZSIsInJlZiI6Imhqc2N5bWZvbHV1cXNsdGZzaHh1Iiwicm9sZSI6ImFub24iLCJpYXQiOjE3ODM1NzQxNzcsImV4cCI6MjA5OTE1MDE3N30.AVaR1LWvnaONa5KIYr4rTteqZ-VgdPOvFgGvLw8f9Zk'; 
    const supabaseClient = supabase.createClient(SUPABASE_URL, SUPABASE_ANON_KEY);

    const studentId = localStorage.getItem('supabase_student_id');
    let gems = 1000;

    let bossHP = 300, maxBossHP = 300;
    let playerHP = 200, maxPlayerHP = 200;
    let selectedCard = null;
    let currentAns = 0;
    let currentTopic = '';

    // 初始化數據
    async function initBattle() {
        if (!studentId) {
            alert('登入失效，回到傳送門！');
            window.location.href = 'index.html';
            return;
        }
        const { data } = await supabaseClient.from('student_progress').select('gems').eq('student_id', studentId).single();
        if (data) {
            gems = data.gems;
            document.getElementById('gemCount').innerText = gems;
        }
    }

    // 選擇手牌，生成隨機 DSE 數學題
    function selectCard(hero, element) {
        selectedCard = hero;
        document.querySelectorAll('.combat-card').forEach(c => c.classList.remove('selected'));
        element.classList.add('selected');

        document.getElementById('combatFeedback').innerText = '';

        if (hero === 'Pascal') {
            currentTopic = 'Permutations_Combinations';
            // 隨機生成 P&C 題目
            const items = Math.floor(Math.random() * 3) + 5; // 5-7
            const select = 3;
            // 題目：從 items 中選 3 個的組合數 C(items, 3)
            currentAns = factorial(items) / (factorial(select) * factorial(items - select));
            document.getElementById('questText').innerText = `【巴斯卡之盾】發動中！城堡防線上需要部署元素藥水。從 ${items} 種不同的水晶中，任選 ${select} 種，一共有多少種不同的【組合 (Combinations)】？`;
        } 
        else if (hero === 'Gauss') {
            currentTopic = 'Expected_Value';
            // 隨機生成期望值題目
            const dmg1 = Math.floor(Math.random() * 2) * 100 + 200; // 200 或 300
            const prob1Percent = Math.floor(Math.random() * 2) * 20 + 60; // 60% 或 80%
            const prob1 = prob1Percent / 100;
            currentAns = dmg1 * prob1;
            document.getElementById('questText').innerText = `【高斯雷擊領域】啟動！陣型有 ${prob1Percent}% 的機率造成 ${dmg1} 點高額雷擊傷害（其餘機率造成 0 傷害），請精算此陣型的【傷害期望值 (Expected Value)】為多少點？`;
        } 
        else if (hero === 'Bayes') {
            currentTopic = 'Conditional_Probability';
            // 隨機生成條件概率題目
            const baseProbPercent = 20;
            const wetProbPercent = 80;
            const probDecimal = wetProbPercent / 100;
            // 題目：已知潮濕狀態下（發生機率100%），連續兩次暴擊的機率
            currentAns = parseFloat((probDecimal * probDecimal * 100).toFixed(1));
            document.getElementById('questText').innerText = `【貝葉斯天雷神罰】發動！已知敵方處於「潮濕狀態」，單次天雷造成暴擊的機率提升至 ${wetProbPercent}%。在此狀態下，你【連續兩次】發動天雷皆造成暴擊的機率是多少%？(四捨五入至小數點後一位)`;
        }
    }

    // 施放技能
    async function castSkill() {
        const studentAns = parseFloat(document.getElementById('battleInput').value);
        const fb = document.getElementById('combatFeedback');

        if (!selectedCard) {
            fb.innerText = '⚠️ 請先選擇一張手牌！';
            fb.style.color = '#ff4d4d';
            return;
        }

        if (isNaN(studentAns)) {
            fb.innerText = '⚠️ 請輸入你的精算答案！';
            fb.style.color = '#ff4d4d';
            return;
        }

        let isCorrect = Math.abs(studentAns - currentAns) <= 0.2;

        if (isCorrect) {
            bossHP -= 100;
            fb.innerText = '⚡【完美精算！】必殺技完美連擊，對混沌獸造成 100 點痛擊！';
            fb.style.color = '#00ffcc';
            
            if (bossHP <= 0) {
                bossHP = 0;
                fb.innerText = '🏆【大捷！】混沌獸已被封印！恭喜獲得 150 寶石！';
                gems += 150;
                document.getElementById('gemCount').innerText = gems;
                await supabaseClient.from('student_progress').update({ gems }).eq('student_id', studentId);
            }
        } else {
            playerHP -= 40;
            fb.innerText = `❌【公式干擾！】計算失誤魔法反噬，防線遭受攻擊，玩家扣除 40 HP！（答案是 ${currentAns}）`;
            fb.style.color = '#ff4d4d';
            if (playerHP <= 0) {
                playerHP = 0;
                fb.innerText = '💀 戰線崩潰！請回到大廳整頓旗鼓，再次精算挑戰！';
            }
        }

        updateBattleUI();
        await updateDSERecordsCloud(isCorrect);
        document.getElementById('battleInput').value = '';
    }

    function updateBattleUI() {
        document.getElementById('bossHP').style.width = `${(bossHP / maxBossHP) * 100}%`;
        document.getElementById('playerHP').style.width = `${(playerHP / maxPlayerHP) * 100}%`;
        document.getElementById('bossName').innerText = `混沌獸 (HP: ${bossHP})`;
    }

    async function updateDSERecordsCloud(isCorrect) {
        try {
            const { data: record } = await supabaseClient.from('dse_records').select('*').eq('student_id', studentId).eq('topic', currentTopic).single();
            if (record) {
                await supabaseClient.from('dse_records').update({
                    total_attempts: record.total_attempts + 1,
                    correct_attempts: record.correct_attempts + (isCorrect ? 1 : 0)
                }).eq('id', record.id);
            } else {
                await supabaseClient.from('dse_records').insert([{
                    student_id: studentId,
                    topic: currentTopic,
                    total_attempts: 1,
                    correct_attempts: isCorrect ? 1 : 0
                }]);
            }
        } catch (e) {
            console.error('上傳成績出錯：', e);
        }
    }

    function factorial(n) {
        if (n === 0 || n === 1) return 1;
        return n * factorial(n - 1);
    }

    initBattle();
</script>
